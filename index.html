function saleTotals(t){ let usd = 0; let iqd = 0; (t.passengers || []).forEach(p => { const amount = Number(p.saleAmount || 0); if(p.saleCurrency === "IQD"){ iqd += Number(p.saleIQD || amount || 0); }else{ usd += Number(p.saleUSD || amount || 0); } }); return { usd: usd, iqd: iqd }; } /* ================================ تسديدات الزبون ================================ */ function customerPaid(t){ let usd = 0; let iqd = 0; let equivalentUSDValue = 0; let equivalentIQDValue = 0; (t.customerPayments || []).forEach(p => { const paymentUSD = Number(p.usd || 0); const paymentIQD = Number(p.iqd || 0); const paymentRate = Number( p.exchangeRate || t.exchangeRate || 1550 ); usd += paymentUSD; iqd += paymentIQD; /* * قيمة التسديد بالدولار */ equivalentUSDValue += paymentUSD + (paymentIQD / paymentRate); /* * قيمة التسديد بالدينار */ equivalentIQDValue += paymentIQD + (paymentUSD * paymentRate); }); return { usd: usd, iqd: iqd, equivalentUSD: equivalentUSDValue, equivalentIQD: equivalentIQDValue }; } /* ================================ تسديدات المورد ================================ */ function supplierPaid(t){ let usd = 0; let iqd = 0; (t.supplierPayments || []).forEach(p => { usd += Number(p.usd || 0); iqd += Number(p.iqd || 0); }); return { usd: usd, iqd: iqd }; } /* ================================ تحويل إلى دولار ================================ */ function equivalentUSD(usd, iqd, rate){ rate = Number(rate || 1550); if(rate <= 0){ rate = 1550; } return Number(usd || 0) + (Number(iqd || 0) / rate); } /* ================================ تحويل إلى دينار ================================ */ function equivalentIQD(usd, iqd, rate){ rate = Number(rate || 1550); if(rate <= 0){ rate = 1550; } return Number(iqd || 0) + (Number(usd || 0) * rate); } /* ================================ ربح العملية بالدينار ================================ */ function operationProfitIQD(t){ const sale = saleTotals(t); const rate = Number(t.exchangeRate || 1550); const saleIQDValue = equivalentIQD( sale.usd, sale.iqd, rate ); const supplierIQDValue = equivalentIQD( Number(t.supplierAmountUSD || 0), Number(t.supplierAmountIQD || 0), rate ); return saleIQDValue - supplierIQDValue; } /* ================================ ربح العملية بالدولار ================================ */ function operationProfitUSD(t){ const rate = Number(t.exchangeRate || 1550); if(rate <= 0){ return 0; } return operationProfitIQD(t) / rate; } /* ================================ حالة ربح العملية ================================ */ function operationProfitStatus(t){ const profit = operationProfitIQD(t); if(profit > 0.01){ return "ربح"; } if(profit < -0.01){ return "خسارة"; } return "تعادل"; } /* ================================ رصيد الزبون ================================ */ function customerBalance(t){ const sale = saleTotals(t); const paid = customerPaid(t); const rate = Number(t.exchangeRate || 1550); /* * نحول البيع كله إلى الدينار */ const saleIQDValue = equivalentIQD( sale.usd, sale.iqd, rate ); /* * التسديدات كلها إلى الدينار * * الدولار يستخدم سعر صرف التسديد * إذا كان محفوظاً معه. */ let paidIQDValue = 0; (t.customerPayments || []).forEach(p => { const paymentUSD = Number(p.usd || 0); const paymentIQD = Number(p.iqd || 0); const paymentRate = Number( p.exchangeRate || rate || 1550 ); paidIQDValue += paymentIQD + (paymentUSD * paymentRate); }); /* * المتبقي بالدينار */ const remainingIQD = Math.max( 0, saleIQDValue - paidIQDValue ); /* * المتبقي بالدولار */ const remainingUSD = rate > 0 ? remainingIQD / rate : 0; return { usd: remainingUSD, iqd: remainingIQD }; } /* ================================ حالة حساب الزبون ================================ */ function customerStatus(t){ const balance = customerBalance(t); if(Number(balance.iqd || 0) > 0.01){ return "متبقي"; } return "مسدد"; } /* ================================ رصيد المورد ================================ */ function supplierBalance(t){ const paid = supplierPaid(t); const rate = Number(t.exchangeRate || 1550); const supplierUSD = Number( t.supplierAmountUSD || 0 ); const supplierIQD = Number( t.supplierAmountIQD || 0 ); /* * إجمالي كلفة المورد بالدينار */ const supplierIQDValue = equivalentIQD( supplierUSD, supplierIQD, rate ); /* * إجمالي ما تم دفعه للمورد بالدينار */ const paidIQDValue = equivalentIQD( paid.usd, paid.iqd, rate ); /* * المتبقي */ const remainingIQD = Math.max( 0, supplierIQDValue - paidIQDValue ); const remainingUSD = rate > 0 ? remainingIQD / rate : 0; return { usd: remainingUSD, iqd: remainingIQD }; } /* ================================ حالة حساب المورد ================================ */ function supplierStatus(t){ const balance = supplierBalance(t); if(Number(balance.iqd || 0) > 0.01){ return "متبقي"; } return "مسدد"; }
