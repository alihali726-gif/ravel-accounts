function saleTotals(t){ let usd = 0; let iqd = 0; (t.passengers || []).forEach(p => { const amount = Number(p.saleAmount || 0); if(p.saleCurrency === "IQD"){ iqd += Number(p.saleIQD || amount || 0); }else{ usd += Number(p.saleUSD || amount || 0); } }); return { usd: usd, iqd: iqd }; } function customerPaid(t){ let usd = 0; let iqd = 0; let equivalentUSDValue = 0; let equivalentIQDValue = 0; (t.customerPayments || []).forEach(p => { const paymentUSD = Number(p.usd || 0); const paymentIQD = Number(p.iqd || 0); const paymentRate = Number( p.exchangeRate || t.exchangeRate || 1550 ); usd += paymentUSD; iqd += paymentIQD; equivalentUSDValue += paymentUSD + (paymentIQD / paymentRate); equivalentIQDValue += paymentIQD + (paymentUSD * paymentRate); }); return { usd: usd, iqd: iqd, equivalentUSD: equivalentUSDValue, equivalentIQD: equivalentIQDValue }; } } function supplierPaid(t){ let usd = 0; let iqd = 0; (t.supplierPayments || []).forEach(p => { usd += Number(p.usd || 0); iqd += Number(p.iqd || 0); }); return { usd: usd, iqd: iqd }; } function equivalentUSD(usd, iqd, rate){ rate = Number(rate || 1550); if(rate <= 0){ rate = 1550; } return Number(usd || 0) + (Number(iqd || 0) / rate); } function equivalentIQD(usd, iqd, rate){ rate = Number(rate || 1550); if(rate <= 0){ rate = 1550; } return Number(iqd || 0) + (Number(usd || 0) * rate); } /* ================================ ربح العملية ================================ */ function operationProfitIQD(t){ const sale = saleTotals(t); const rate = Number(t.exchangeRate || 1550); const saleIQDValue = equivalentIQD( sale.usd, sale.iqd, rate ); const supplierIQDValue = equivalentIQD( Number(t.supplierAmountUSD || 0), Number(t.supplierAmountIQD || 0), rate ); return saleIQDValue - supplierIQDValue; } function operationProfitUSD(t){ const rate = Number(t.exchangeRate || 1550); if(rate <= 0){ return 0; } return operationProfitIQD(t) / rate; } function operationProfitStatus(t){ const profit = operationProfitIQD(t); if(profit > 0.01){ return "ربح"; } if(profit < -0.01){ return "خسارة"; } return "تعادل"; } /* ================================ رصيد الزبون ================================ */ function customerBalance(t){ const sale = saleTotals(t); const paid = customerPaid(t); const rate = Number(t.exchangeRate || 1550); /* * قيمة البيع بالدولار حسب سعر صرف العملية */ const saleUSDValue = equivalentUSD( sale.usd, sale.iqd, rate ); /* * قيمة البيع بالدينار حسب سعر صرف العملية */ const saleIQDValue = equivalentIQD( sale.usd, sale.iqd, rate ); /* * التسديدات: * إذا كان التسديد بالدولار، يتم تحويله * حسب سعر صرف التسديد نفسه. * * إذا كان التسديد بالدينار، يبقى كما هو. */ const paidUSDValue = Number( paid.equivalentUSD || 0 ); const paidIQDValue = Number( paid.equivalentIQD || 0 ); /* * الرصيد المتبقي */ return { usd: Math.max( 0, saleUSDValue - paidUSDValue ), iqd: Math.max( 0, saleIQDValue - paidIQDValue ) }; } const paidUSDValue = equivalentUSD( paid.usd, paid.iqd, rate ); const saleIQDValue = equivalentIQD( sale.usd, sale.iqd, rate ); const paidIQDValue = equivalentIQD( paid.usd, paid.iqd, rate ); return { usd: Math.max( 0, saleUSDValue - paidUSDValue ), iqd: Math.max( 0, saleIQDValue - paidIQDValue ) }; } /* ================================ حالة حساب الزبون ================================ */ function customerStatus(t){ const balance = customerBalance(t); const rate = Number(t.exchangeRate || 1550); const remainingUSD = equivalentUSD( balance.usd, balance.iqd, rate ); if(remainingUSD > 0.01){ return "متبقي"; } return "مسدد"; } /* ================================ رصيد المورد ================================ */ function supplierBalance(t){ const paid = supplierPaid(t); const rate = Number(t.exchangeRate || 1550); const supplierUSD = Number( t.supplierAmountUSD || 0 ); const supplierIQD = Number( t.supplierAmountIQD || 0 ); const supplierUSDValue = equivalentUSD( supplierUSD, supplierIQD, rate ); const paidUSDValue = equivalentUSD( paid.usd, paid.iqd, rate ); const supplierIQDValue = equivalentIQD( supplierUSD, supplierIQD, rate ); const paidIQDValue = equivalentIQD( paid.usd, paid.iqd, rate ); return { usd: Math.max( 0, supplierUSDValue - paidUSDValue ), iqd: Math.max( 0, supplierIQDValue - paidIQDValue ) }; } /* ================================ حالة حساب المورد ================================ */ function supplierStatus(t){ const balance = supplierBalance(t); const rate = Number(t.exchangeRate || 1550); const remainingUSD = equivalentUSD( balance.usd, balance.iqd, rate ); if(remainingUSD > 0.01){ return "متبقي"; } return "مسدد"; }
